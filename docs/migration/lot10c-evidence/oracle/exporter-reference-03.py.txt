def encoded(value):
    return json.dumps(value, ensure_ascii=False, allow_nan=False, sort_keys=True, separators=(',', ':')).encode()


def plain(value):
    if hasattr(value, 'model_dump'): return plain(value.model_dump())
    if hasattr(value, 'items'): return {str(k): plain(v) for k, v in value.items()}
    if isinstance(value, (list, tuple)): return [plain(v) for v in value]
    if isinstance(value, (set, frozenset)): return [plain(v) for v in sorted(value)]
    return value


def verify_sources():
    manifest = json.loads((SOURCE.parent / 'manifest.json').read_text())
    assert manifest['commit'] == COMMIT and len(manifest['files']) == 1786
    for name, record in manifest['files'].items():
        raw = (SOURCE / name).read_bytes()
        assert len(raw) == record['bytes'] and hashlib.sha256(raw).hexdigest() == record['sha256'], name
    return manifest


def capture(call):
    try: return {'value': plain(call())}
    except Exception as error:
        result = {'name': type(error).__name__, 'message': str(error)}
        if hasattr(error, 'cause'): result['cause'] = error.cause
        if hasattr(error, 'frontier'): result['frontier'] = plain(error.frontier)
        return {'error': result}


def box(rows, left, top, right, bottom, color, outline=False):
    for y in range(top, bottom + 1):
        for x in range(left, right + 1):
            if not outline or x in (left, right) or y in (top, bottom): rows[y][x] = color


def inward_coordinates(n):
    # Independent construction by sorted rings and perimeter position. This is
    # an input layout, not a copied expected queue or a native measurement.
    points = [(x, y) for y in range(n) for x in range(n)]
    def position(point):
        x, y = point; ring = min(x, y, n - 1 - x, n - 1 - y)
        left = top = ring; right = bottom = n - 1 - ring; edge = right - left
        if y == top: offset = x - left
        elif x == right: offset = edge + y - top
        elif y == bottom: offset = 2 * edge + right - x
        else: offset = 3 * edge + bottom - y
        return ring, offset
    return sorted(points, key=position)


def raster(queues=None, enemies=None, phase='COMBAT', quota=1, quotas=None,
           palette=((2, 3), (10, 11), (12, 13)), vertical=False, shift=(0, 0),
           link_directions=None, fill_links=False):
    queues = [[0], []] if queues is None else deepcopy(queues)
    enemies = [[1], [2]] if enemies is None else deepcopy(enemies)
    assert len(queues) == len(enemies) and len(queues) >= 2
    count = len(queues); quotas = [1] * count if quotas is None else list(quotas)
    dx, dy = shift; top = 18 + dy
    positions = [(2 + dx + (0 if vertical else 28 * i), top + (18 * i if vertical else 0)) for i in range(count)]
    room_bottom = max(y + 12 for x, y in positions); button_top = room_bottom + 12
    width = max(58 + dx, max(x + 23 for x, y in positions)); height = button_top + 13
    rows = [[0] * width for _ in range(height)]
    for index, pair in enumerate(palette):
        y = 1 + 5 * index + dy
        following = palette[(index + 1) % len(palette)]
        for x, value in zip((1 + dx, 5 + dx), pair): box(rows, x, y, x + 2, y + 2, value)
        for x, value in zip((14 + dx, 18 + dx), following): box(rows, x, y, x + 1, y + 1, value)
        rows[y][10 + dx] = 5; rows[y + 1][11 + dx] = 5; rows[y + 2][10 + dx] = 5
    for index, ((left, y), queue, enemy) in enumerate(zip(positions, queues, enemies)):
        box(rows, left, y, left + 20, y + 12, 7, outline=True)
        xs_left = [left + 3, left + 5, left + 7]; xs_right = [left + 13, left + 15, left + 17]
        ys = [y + 3, y + 5, y + 7]
        for yy in ys:
            for xx in xs_left + xs_right: rows[yy][xx] = 1
        for queue_values, xs, camp in ((queue, list(reversed(xs_left)), 0), (enemy, xs_right, 1)):
            for typ, (u, v) in zip(queue_values, inward_coordinates(3)):
                rows[ys[v] - 1][xs[u] - 1] = palette[typ][camp]
            if queue_values: rows[ys[0]][xs[0]] = 14
        for token in range(quotas[index]): rows[y + 10][left + 3 + 2 * token] = 6
    for i in range(count - 1):
        x, y = positions[i]; direction = 1 if link_directions is None else link_directions[i]
        if vertical:
            center = (x + 10, y + 15)
            tip = (center[0], center[1] + direction)
            wings = [(center[0] - 1, center[1]), (center[0] + 1, center[1])]
        else:
            center = (x + 24, y + 6)
            tip = (center[0] + direction, center[1])
            wings = [(center[0], center[1] - 1), (center[0], center[1] + 1)]
        for xx, yy in [tip, *wings, *([center] if fill_links else [])]: rows[yy][xx] = 4
    if phase == 'RECRUIT':
        left, y = 2 + dx, room_bottom + 4
        box(rows, left, y, left + max(10, 2 * quota + 2), y + 4, 7, outline=True)
        for token in range(quota): box(rows, left + 2 + 2 * token, y + 1, left + 2 + 2 * token, y + 3, 7)
    for index, pair in enumerate(palette):
        x = 2 + dx + index * 10
        box(rows, x, button_top, x + 5, button_top + 9, 7 if phase == 'RECRUIT' else 8, outline=True)
        rows[button_top + 4][x + 2] = pair[0]
    return rows
